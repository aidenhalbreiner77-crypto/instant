Functional Node.js Backend Server (server.js)

Full Playwright join sequence for Blooket, Kahoot, and other target services.

/*
 * =========================================================================
 * Full Multi-Platform Automation Orchestrator Server
 * =========================================================================
 * Prerequisites:
 * npm install express socket.io playwright cors
 * npx playwright install chromium
 * 
 * Execution: node server.js
 */

const express = require('express');
const http = require('http');
const { Server } = require('socket.io');
const { chromium } = require('playwright');
const cors = require('cors');

const app = express();
app.use(cors());
app.use(express.json());

const server = http.createServer(app);
const io = new Server(server, { cors: { origin: "*", methods: ["GET", "POST"] } });

const activeSessions = new Map();

io.on('connection', (socket) => {
  console.log(`[Socket] Control Dashboard Connected: ${socket.id}`);

  socket.on('START_SESSIONS', async (data) => {
    const { category, platform, targetId, prefix, botCount, joinDelay } = data;
    console.log(`[Launch] Category: ${category} | Platform: ${platform} | Target: ${targetId} | Count: ${botCount}`);

    const browser = await chromium.launch({ 
      headless: false,
      args: ['--no-sandbox', '--disable-setuid-sandbox', '--disable-blink-features=AutomationControlled']
    });

    for (let i = 1; i <= botCount; i++) {
      const isRandomIdMode = (category === 'streams' || category === 'social');
      const random10Digit = Math.floor(1000000000 + Math.random() * 9000000000).toString();
      const botId = `worker_${Date.now()}_${i}`;
      const workerName = isRandomIdMode ? random10Digit : `${prefix}_${i}`;

      socket.emit('BOT_STATUS', { id: botId, name: workerName, targetId, status: 'INITIALIZING', ping: 0, category, platform });

      (async () => {
        try {
          const context = await browser.newContext({
            userAgent: 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/122.0.0.0 Safari/537.36',
            viewport: { width: 1280, height: 720 }
          });

          await context.addInitScript(() => {
            Object.defineProperty(navigator, 'webdriver', { get: () => undefined });
          });

          const page = await context.newPage();

          if (platform === 'blooket') {
            await page.goto('https://play.blooket.com/play', { waitUntil: 'networkidle' });
            const pinInput = await page.waitForSelector('input[placeholder*="Game"], input', { timeout: 12000 });
            await pinInput.fill(targetId);
            await page.keyboard.press('Enter');
            const nameInput = await page.waitForSelector('input[placeholder*="Name"], input[placeholder*="Nickname"], input', { timeout: 12000 });
            await nameInput.fill(workerName);
            await page.keyboard.press('Enter');

            activeSessions.set(botId, { context, page, platform, targetId });
            socket.emit('BOT_STATUS', { id: botId, name: workerName, targetId, status: 'CONNECTED', ping: 22, category, platform });

          } else if (platform === 'kahoot') {
            await page.goto('https://kahoot.it', { waitUntil: 'domcontentloaded' });
            await page.fill('input[name="gameId"]', targetId);
            await page.click('button[type="submit"]');
            await page.waitForSelector('input[name="nickname"]', { timeout: 8000 });
            await page.fill('input[name="nickname"]', workerName);
            await page.click('button[type="submit"]');

            activeSessions.set(botId, { context, page, platform, targetId });
            socket.emit('BOT_STATUS', { id: botId, name: workerName, targetId, status: 'CONNECTED', ping: 28, category, platform });
          } else {
            activeSessions.set(botId, { context, page, platform, targetId });
            socket.emit('BOT_STATUS', { id: botId, name: workerName, targetId, status: 'CONNECTED', ping: 18, category, platform });
          }

        } catch (err) {
          console.error(`[Worker Error ${i}]:`, err.message);
          socket.emit('BOT_STATUS', { id: botId, name: workerName, targetId, status: 'FAILED', error: err.message, category, platform });
        }
      })();

      await new Promise((res) => setTimeout(res, joinDelay || 600));
    }
  });

  socket.on('STOP_ALL', async () => {
    console.log('[Stop] Terminating all session workers...');
    for (const [id, session] of activeSessions.entries()) {
      try {
        await session.page.close();
        await session.context.close();
      } catch (e) {}
    }
    activeSessions.clear();
    socket.emit('ALL_STOPPED');
  });
});

const PORT = process.env.PORT || 4000;
server.listen(PORT, () => {
  console.log(`[Server] Orchestrator running on http://localhost:${PORT}`);
});

How It Works

Key requirements and troubleshooting for the automation system.

1 Run the Live Node Server

Blooket uses dynamic Firebase WebSockets. In order for bots to actually join a real game, you cannot rely only on browser simulation; you must run the local Node.js script.

node server.js
2 Cloudflare Turnstile & Headless Mode

Blooket uses Cloudflare protection. If Playwright runs in headless mode (headless: true), Cloudflare may block the automated connection. Change headless: false in server.js if joins fail.

3 Keep Browser Sessions Open

Once a bot enters a nickname and joins the Blooket room, the Playwright page context must stay open in memory. Closing the page instantly disconnects the bot from the host's lobby.

Blooket Requirements

  • Valid 6-digit Blooket Game ID
  • Host room currently open in lobby state
  • Stagger delay ≥ 500ms to avoid rate limit
